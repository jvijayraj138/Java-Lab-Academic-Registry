Week 12 — Character Streams and Multithreaded Programming

Complete collection of 12 Java programs

How to use: Select a program to expand it, then click Copy code. Save each Java program in a separate .java file using the same name as its public class. For file-handling programs, create the required text files in the program's working directory.

Programs

1. Keyboard Input Using Reader and Writer

Reads characters from the keyboard and writes them to the console.

Java
import java.io.*;

public class KeyboardToConsole {
    public static void main(String[] args) {
        System.out.println("Enter text (press Ctrl+Z then Enter to stop on Windows):");
        try {
            Reader reader = new InputStreamReader(System.in);
            Writer writer = new OutputStreamWriter(System.out);
            int ch;
            while ((ch = reader.read()) != -1) {
                writer.write(ch);
                writer.flush();
            }
        } catch (IOException e) {
            System.out.println("I/O Error: " + e.getMessage());
        }
    }
}

Java
import java.io.*;

4. Copy One Text File to Another

Create source.txt first. The copied content will be saved in destination.txt.

Java
import java.io.*;

public class CopyFileDemo {
    public static void main(String[] args) {
        try (FileReader reader = new FileReader("source.txt");
             FileWriter writer = new FileWriter("destination.txt")) {
            int ch;
            while ((ch = reader.read()) != -1) {
                writer.write(ch);
            }
            System.out.println("File copied successfully.");
        } catch (IOException e) {
            System.out.println("Error copying file: " + e.getMessage());
        }
    }
}
5. Count Characters, Words, and Lines in a File

Uses input.txt. Counts characters, whitespace-separated words, and text lines.

Java
import java.io.*;

public class FileCounterDemo {
    public static void main(String[] args) {
        int characters = 0, words = 0, lines = 0;
        boolean inWord = false;
        boolean hasContent = false;

        try (FileReader reader = new FileReader("input.txt")) {
            int value;
            while ((value = reader.read()) != -1) {
                char ch = (char) value;
                characters++;
                hasContent = true;

                if (ch == '\\n') {
                    lines++;
                }

                if (Character.isWhitespace(ch)) {
                    inWord = false;
                } else if (!inWord) {
                    words++;
                    inWord = true;
                }
            }

            if (hasContent && (characters == 0 || lines == 0 ||
                readLastCharacterIsNotNewline("input.txt"))) {
                lines++;
            }

            System.out.println("Characters: " + characters);
            System.out.println("Words: " + words);
            System.out.println("Lines: " + lines);
        } catch (IOException e) {
            System.out.println("Error reading file: " + e.getMessage());
        }
    }

    private static boolean readLastCharacterIsNotNewline(String file) throws IOException {
        try (RandomAccessFile raf = new RandomAccessFile(file, "r")) {
            if (raf.length() == 0) return false;
            raf.seek(raf.length() - 1);
            return raf.read() != '\\n';
        }
    }
}

7. Create a Thread by Extending Thread

The start() method begins a new thread and calls run() internally.

Java
class MyThread extends Thread {
    public void run() {
        for (int i = 1; i <= 5; i++) {
            System.out.println("Child thread: " + i);
        }
    }
}

public class ThreadClassDemo {
    public static void main(String[] args) {
        MyThread t = new MyThread();

    }

10. Demonstrate isAlive()

isAlive() returns true while a thread has started and has not yet terminated.

Java
public class IsAliveDemo {
    public static void main(String[] args) {
        Thread t = new Thread(() -> {
            for (int i = 1; i <= 3; i++) {
                System.out.println("Child thread: " + i);
                try {
                    Thread.sleep(300);
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                }
            }
        });

        System.out.println("Before start: " + t.isAlive());
        t.start();
        System.out.println("After start: " + t.isAlive());

        try {
            t.join();
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        System.out.println("After completion: " + t.isAlive());
    }
}
11. Demonstrate join()

join() makes the main thread wait until the worker thread finishes.

Java
class JoinWorker extends Thread {
    public void run() {
        for (int i = 1; i <= 5; i++) {
            System.out.println("Worker: " + i);
            try {
                Thread.sleep(200);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                return;
            }
        }
    }
}

public class JoinDemo {
    public static void main(String[] args) {
        JoinWorker t = new JoinWorker();
        t.start();

        try {
            t.join();
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        System.out.println("Worker thread has completed.");
        System.out.println("Main thread continues.");
    }
}
12. Use isAlive() and join() with Multiple Threads

Uses isAlive() to check thread status and join() to wait for both threads.

Java
class TaskThread extends Thread {
    TaskThread(String name) {
        super(name);
    }

    public void run() {
        for (int i = 1; i <= 3; i++) {
            System.out.println(getName() + " : " + i);
            try {
                Thread.sleep(200);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
                return;
            }
        }
    }
}

public class ThreadControlDemo {
    public static void main(String[] args) {
        TaskThread t1 = new TaskThread("Thread-1");
        TaskThread t2 = new TaskThread("Thread-2");

        t1.start();
        t2.start();

        System.out.println("Thread-1 alive: " + t1.isAlive());
        System.out.println("Thread-2 alive: " + t2.isAlive());

        try {
            t1.join();
            t2.join();
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
        }

        System.out.println("Thread-1 alive after join: " + t1.isAlive());
        System.out.println("Thread-2 alive after join: " + t2.isAlive());
        System.out.println("Both threads have completed.");
    }
}
