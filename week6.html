Program 2.2: Array of Student Objects

import java.util.Scanner;
class Student {
    int rollNumber;
    String studentName;
    String branch;
    double cgpa;
    void displayStudent() {
        System.out.println("Roll Number : " + rollNumber);
        System.out.println("Name        : " + studentName);
        System.out.println("Branch      : " + branch);
        System.out.println("CGPA        : " + cgpa);
        System.out.println();
    }
}
public class StudentArray {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        Student[] students = new Student[5];
        for (int i = 0; i < students.length; i++) {
            students[i] = new Student();
            System.out.println("Enter details of Student " + (i + 1));
            System.out.print("Roll Number: ");
            students[i].rollNumber = input.nextInt();
            input.nextLine();
            System.out.print("Name: ");
            students[i].studentName = input.nextLine();
            System.out.print("Branch: ");
            students[i].branch = input.nextLine();
            System.out.print("CGPA: ");
            students[i].cgpa = input.nextDouble();
            System.out.println();
        }
        System.out.println("\n------ Student Details ------");
        for (int i = 0; i < students.length; i++) {
            System.out.println("Student " + (i + 1));
            students[i].displayStudent();
        }
        input.close();
    }
}

Program 4.1: Calculator Using Methods

import java.util.Scanner;
class Calculator {
    void addition(double num1, double num2) {
        System.out.println("Addition = " + (num1 + num2));
    }
    void subtraction(double num1, double num2) {
        System.out.println("Subtraction = " + (num1 - num2));
    }
    void multiplication(double num1, double num2) {
        System.out.println("Multiplication = " + (num1 * num2));
    }
    void division(double num1, double num2) {
        if (num2 != 0) {
            System.out.println("Division = " + (num1 / num2));
        } else {
            System.out.println("Division by zero is not possible.");
        }
    }
}
public class CalculatorMethods {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        Calculator calc = new Calculator();
        System.out.print("Enter First Number: ");
        double firstValue = input.nextDouble();
        System.out.print("Enter Second Number: ");
        double secondValue = input.nextDouble();
        System.out.println();
        calc.addition(firstValue, secondValue);
        calc.subtraction(firstValue, secondValue);
        calc.multiplication(firstValue, secondValue);
        calc.division(firstValue, secondValue);
        input.close();
    }
}

Program 5.2: Constructor Overloading

class BankAccount {
    int accountNumber;
    String accountHolder;
    double balance;
    // Default Constructor
    BankAccount() {
        accountNumber = 500101;
        accountHolder = "Sowmya";
        balance = 0.0;
    }
    // Constructor with two parameters
    BankAccount(int accNo, String holderName) {
        accountNumber = accNo;
        accountHolder = holderName;
        balance = 0.0;
    }
    // Constructor with three parameters
    BankAccount(int accNo, String holderName, double amount) {
        accountNumber = accNo;
        accountHolder = holderName;
        balance = amount;
    }
    void displayAccount() {
        System.out.println("Account Number : " + accountNumber);
        System.out.println("Account Holder : " + accountHolder);
        System.out.println("Balance        : ₹" + balance);
        System.out.println();
    }
}
public class ConstructorOverloading {
    public static void main(String[] args) {
        BankAccount accountOne = new BankAccount();
        BankAccount accountTwo = new BankAccount(500102, "Nikhil");
        BankAccount accountThree = new BankAccount(500103, "Harika", 25000.50);
        System.out.println("Default Constructor");
        accountOne.displayAccount();
        System.out.println("Two-Parameter Constructor");
        accountTwo.displayAccount();
        System.out.println("Three-Parameter Constructor");
        accountThree.displayAccount();
    }
}

class Mobile {

    }
    protected void finalize() {
        System.out.println(brand + " object is garbage collected.");

    public static void main(String[] args) {
        // Method 1: Null Assignment
        Mobile phone1 = new Mobile("Samsung");

class Arithmetic {

    int add(int a, int b) {

class BankAccount {
    String accountHolder;
    double balance;
    BankAccount(String accountHolder, double balance) {
        this.accountHolder = accountHolder;
        this.balance = balance;
    }

        return account;

        System.out.println("Updated Balance: " + balance);

    }

Program 11.2: Static Methods

class MathUtility {
    static int square(int n) {
        return n * n;
    }
    static int cube(int n) {
        return n * n * n;
    }
    static int factorial(int n) {
        int fact = 1;
        for (int i = 1; i <= n; i++) {
            fact = fact * i;
        }
        return fact;
    }
    public static void main(String[] args) {
        int number = 6;
        System.out.println("Number: " + number);
        System.out.println("Square: " + square(number));
        System.out.println("Cube: " + cube(number));
        System.out.println("Factorial: " + factorial(number));
    }
}
