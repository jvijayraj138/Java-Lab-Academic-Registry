Experiment 1: Single Inheritance

Aim: To demonstrate single inheritance using Person and Student classes.

import java.util.Scanner;

class Person {
    String name;
    int age;

    void getPersonDetails(Scanner sc) {
        System.out.print("Enter name: ");
        name = sc.nextLine();

        System.out.print("Enter age: ");
        age = sc.nextInt();
    }
}

class Student extends Person {
    int rollNo;
    String branch;

    void getStudentDetails(Scanner sc) {
        System.out.print("Enter roll number: ");
        rollNo = sc.nextInt();
        sc.nextLine();

        System.out.print("Enter branch: ");
        branch = sc.nextLine();
    }

    void display() {
        System.out.println("\n--- Student Details ---");
        System.out.println("Name       : " + name);
        System.out.println("Age        : " + age);
        System.out.println("Roll Number: " + rollNo);
        System.out.println("Branch     : " + branch);
    }
}

public class SingleInheritance {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        Student s = new Student();
        s.getPersonDetails(sc);
        s.getStudentDetails(sc);
        s.display();

        sc.close();
    }
}
Sample Output: 
Enter name: Ravi
Enter age: 19
Enter roll number: 101
Enter branch: CSE

--- Student Details ---
Name       : Ravi
Age        : 19
Roll Number: 101
Branch     : CSE

Experiment 2: Using super Keyword

Aim: To use super to access superclass variables and methods and invoke its constructor.

import java.util.Scanner;

class Vehicle {
    int speed;

    Vehicle(int speed) {
        this.speed = speed;
    }

    void display() {
        System.out.println("Vehicle Speed: " + speed + " km/h");
    }
}

class Car extends Vehicle {
    String model;

    Car(int speed, String model) {
        super(speed);
        this.model = model;
    }

    @Override
    void display() {
        super.display();
        System.out.println("Car Model : " + model);
        System.out.println("Super Speed : " + super.speed + " km/h");
    }
}

public class SuperKeywordDemo {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter car speed: ");
        int speed = sc.nextInt();
        sc.nextLine();

        System.out.print("Enter car model: ");
        String model = sc.nextLine();

        Car c = new Car(speed, model);
        c.display();

        sc.close();
    }
}
Sample Output: 
Enter car speed: 120
Enter car model: Honda

Vehicle Speed: 120 km/h
Car Model : Honda
Super Speed : 120 km/h

Experiment 4: Method Overriding

Aim: To demonstrate method overriding using Animal, Dog and Cat classes.

import java.util.Scanner;

class Animal {
    void sound() {
        System.out.println("Animal makes a sound");
    }
}

class Dog extends Animal {
    @Override
    void sound() {
        System.out.println("Dog says: Woof Woof");
    }
}

class Cat extends Animal {
    @Override
    void sound() {
        System.out.println("Cat says: Meow Meow");
    }
}

public class MethodOverriding {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter animal (dog/cat): ");
        String choice = sc.nextLine();

        Animal animal;

        if (choice.equalsIgnoreCase("dog")) {
            animal = new Dog();
        } else if (choice.equalsIgnoreCase("cat")) {
            animal = new Cat();
        } else {
            System.out.println("Invalid animal.");
            sc.close();
            return;
        }

        animal.sound();
        sc.close();
    }
}
Sample Output: 
Enter animal (dog/cat): dog
Dog says: Woof Woof

Experiment 7: Multilevel Inheritance for Salary Calculation

Aim: To calculate total salary using Employee, Developer and SeniorDeveloper classes.

import java.util.Scanner;

class Employee {
    double basicSalary;

    void getBasicSalary(Scanner sc) {
        System.out.print("Enter basic salary: ");
        basicSalary = sc.nextDouble();
    }
}

class Developer extends Employee {
    double programmingAllowance;

    void getProgrammingAllowance(Scanner sc) {
        System.out.print("Enter programming allowance: ");
        programmingAllowance = sc.nextDouble();
    }
}

class SeniorDeveloper extends Developer {
    double projectAllowance;

    void getProjectAllowance(Scanner sc) {
        System.out.print("Enter project allowance: ");
        projectAllowance = sc.nextDouble();
    }

    void displaySalary() {
        double total = basicSalary
                     + programmingAllowance
                     + projectAllowance;

        System.out.println("\n--- Salary Details ---");
        System.out.println("Basic Salary: " + basicSalary);
        System.out.println("Programming Allowance: "
                           + programmingAllowance);
        System.out.println("Project Allowance: " + projectAllowance);
        System.out.println("Total Salary: " + total);
    }
}

public class SalaryCalculation {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        SeniorDeveloper sd = new SeniorDeveloper();

        sd.getBasicSalary(sc);
        sd.getProgrammingAllowance(sc);
        sd.getProjectAllowance(sc);
        sd.displaySalary();

        sc.close();
    }
}

Experiment 8: Dynamic Method Dispatch for Bank Accounts

Aim: To calculate interest for savings and current accounts using dynamic method dispatch.

import java.util.Scanner;

class BankAccount {
    double balance;

    BankAccount(double balance) {
        this.balance = balance;
    }

    void calculateInterest() {
        System.out.println("General bank interest calculation");
    }
}

class SavingsAccount extends BankAccount {
    SavingsAccount(double balance) {
        super(balance);
    }

    @Override
    void calculateInterest() {
        double interest = balance * 0.04;
        System.out.println("Savings Account Interest: " + interest);
    }
}

class CurrentAccount extends BankAccount {
    CurrentAccount(double balance) {
        super(balance);
    }

    @Override
    void calculateInterest() {
        double interest = balance * 0.02;
        System.out.println("Current Account Interest: " + interest);
    }
}

public class BankDynamicDispatch {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter account type (savings/current): ");
        String type = sc.nextLine();

        System.out.print("Enter balance: ");
        double balance = sc.nextDouble();

        BankAccount account;

        if (type.equalsIgnoreCase("savings")) {
            account = new SavingsAccount(balance);
        } else if (type.equalsIgnoreCase("current")) {
            account = new CurrentAccount(balance);
        } else {
            System.out.println("Invalid account type.");
            sc.close();
            return;
        }

        account.calculateInterest();
        sc.close();
    }
}

Experiment 9: Constructor Execution in Multilevel Inheritance

Aim: To demonstrate the order of constructor execution in Person, Student and GraduateStudent classes.

import java.util.Scanner;

class Person {
    Person(String name) {
        System.out.println("Person constructor executed");

        sc.close();
    }

Experiment 10: Banking Application Using Inheritance

Aim: To demonstrate deposit, withdrawal, interest calculation and method overriding using bank account classes.

import java.util.Scanner;

class BankAccount {
    String accountNumber;
    double balance;

    BankAccount(String accountNumber, double balance) {
        this.accountNumber = accountNumber;
        this.balance = balance;
    }

    void deposit(double amount) {
        balance += amount;
        System.out.println("Deposited: " + amount);
    }

    void withdraw(double amount) {
        if (amount <= balance) {
            balance -= amount;
            System.out.println("Withdrawn: " + amount);
        } else {
            System.out.println("Insufficient balance.");
        }
    }

    void calculateInterest() {
        System.out.println("General interest calculation");
    }

    void displayBalance() {
        System.out.println("Account Number: " + accountNumber);
        System.out.println("Current Balance: " + balance);
    }
}

class SavingsAccount extends BankAccount {
    SavingsAccount(String accountNumber, double balance) {
        super(accountNumber, balance);
    }

    @Override
    void calculateInterest() {
        double interest = balance * 0.05;
        System.out.println("Savings Interest: " + interest);
    }
}

class CurrentAccount extends BankAccount {
    CurrentAccount(String accountNumber, double balance) {
        super(accountNumber, balance);
    }

    @Override
    void calculateInterest() {
        double interest = balance * 0.02;
        System.out.println("Current Account Interest: " + interest);
    }
}

public class BankingApplication {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        System.out.print("Enter account type (savings/current): ");
        String type = sc.nextLine();

        System.out.print("Enter account number: ");
        String accountNumber = sc.nextLine();

        System.out.print("Enter initial balance: ");
        double balance = sc.nextDouble();

        BankAccount account;

        if (type.equalsIgnoreCase("savings")) {
            account = new SavingsAccount(accountNumber, balance);
        } else if (type.equalsIgnoreCase("current")) {
            account = new CurrentAccount(accountNumber, balance);
        } else {
            System.out.println("Invalid account type.");
            sc.close();
            return;
        }

        System.out.print("Enter deposit amount: ");
        double deposit = sc.nextDouble();
        account.deposit(deposit);

        System.out.print("Enter withdrawal amount: ");
        double withdrawal = sc.nextDouble();
        account.withdraw(withdrawal);

        System.out.println();
        account.calculateInterest();
        account.displayBalance();

        sc.close();
    }
}
